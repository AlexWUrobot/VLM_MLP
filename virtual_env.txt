# The previous example used placeholder paths like /path/to/InternVideo.
# Replace those with your real local paths before running the script.

# 1) Clone InternVideo somewhere on disk.
git clone https://github.com/OpenGVLab/InternVideo.git ~/InternVideo

# 2) Activate this project environment first, then install the extra dependencies
# required by InternVideo2 if they are not already present.
pip install timm einops decord sentencepiece

# 3) Download the InternVideo2 checkpoints you want to use.
# Then point the script at the real files, for example:
export INTERNVIDEO2_ROOT=~/InternVideo
export INTERNVIDEO2_VISION_PRETRAINED=/absolute/path/to/internvideo2_1b_patch14_224.pth
# Optional full stage-2 checkpoint if you have one:
# export INTERNVIDEO2_PRETRAINED_PATH=/absolute/path/to/internvideo2_stage2.pt

# 4) Run whole-video quality check + zero-shot labeling.

  --video short_videos/raw/come/come_0.mp4 \
  --labels come,wave,stop,idle,talk_phone,play_phone,thumb_up \
  --label-definitions-json label_definitions.json \
  --internvideo2-root "$INTERNVIDEO2_ROOT" \
  --internvideo2-vision-pretrained "$INTERNVIDEO2_VISION_PRETRAINED" \
  --output-json qc_and_label_internvideo2.json

# If you also have the stage-2 checkpoint, add:
#   --internvideo2-pretrained-path "$INTERNVIDEO2_PRETRAINED_PATH"

# - "Could not find InternVideo2/multi_modality": --internvideo2-root still points

# - "Label definitions file not found": create label_definitions.json in this repo.
# - Model init failures usually mean missing InternVideo dependencies or wrong
